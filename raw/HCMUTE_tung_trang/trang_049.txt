4. Cho biết kết quả khi thực thi chương trình sau:
#include <stdio.h>
int main(void)
{
    int a = 5, c;
    c = a;      
    printf(“c = %d\n”, c); // Kết quả in:......
    c += a;     
    printf(“c = %d\n”, c); // Kết quả in:......
    c -= a;     
    printf(“c = %d\n”, c); // Kết quả in:......
    c *= a;     
    printf(“c = %d\n”, c); // Kết quả in:......
    c /= a;     
    printf(“c = %d\n”, c); // Kết quả in:......
    c %= a;     
    printf(“c = %d\n”, c); // Kết quả in:......
    return 0;
}
