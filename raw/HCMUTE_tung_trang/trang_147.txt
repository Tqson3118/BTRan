void Armstrong (int n)
{
    int s1, s, n1;
    s = 0;
    n1=n;
    while(n1!=0)  
    {  
        s1 = n1 % 10;   
        s += s1 * s1 * s1; 
        n1 = n1/10;  
    }
   if (n == s) printf(“ %d la So armstrong” ,n);

}
