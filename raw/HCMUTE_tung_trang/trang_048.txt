#include <stdio.h>
int main(void)
{
   int a = 10, b = 100;
   float c = 10.5, d = 100.5;
   printf(“++a = %d \n”, ++a); // Kết quả in:.......
   printf(“--b = %d \n”, --b); // Kết quả in:.......
   printf(“++c = %f \n”, ++c); // Kết quả in:.......
   printf(“--d = %f \n”, --d); // Kết quả in:.......
   return 0;
}
