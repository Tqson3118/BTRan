#define INTERGER
#ifdef INTERGER
int timmax(int a, int b)
{
  return a>b ? a:b ;
} 
#else
float timmax(float a, float b)
{
	 return a>b ? a:b ; 
} 
#endif
void main (void)
{
	 int a = 5, b = 9 ; 
	 int max = timmax(a,b);
	 printf (“max = %d”,max);
