2 #include <math.h>  
3 int main() {  
4   unsigned n, k;  
5   double S = 1.0;

7   scanf( "%u", &n );  
8   for ( k = 2; k <= n; ++k )  
9     S = pow( k + S, 1.0 / ( k + 1 ) );

11   return 0;
