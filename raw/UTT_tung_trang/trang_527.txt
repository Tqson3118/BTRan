139  void sapxep(SV a[], int n){ 
140     //Sap xep theo DTB tang dan 
141     SV tmp; 
142     for(int i = 0;i < n;++i){ 
143         for(int j = i+1; j < n;++j){ 
144             if(a[i].dtb > a[j].dtb){ 
145                 tmp = a[i]; 
146                 a[i] = a[j]; 
147                 a[j] = tmp; 
148             } 
149         } 
150     } 
151 }
