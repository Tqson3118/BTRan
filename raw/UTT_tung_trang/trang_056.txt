Xem xét đoạn chƣơng trình sau đây: 
1 #include <stdio.h> 
2 int main () { 
3    /* Khai bao bien cuc bo: */ 
4    int a, b; 
5    /* Khoi tao gia tri */ 
6    a = 20; 
7    b = 7; 
8    printf("% d + % d  = %d\n", a, b, a + b); 
9    printf("% d - % d  = %d\n", a, b, a - b); 
10    printf("% d * % d  = %d\n", a, b, a * b); 
11    printf("% d / % d  = %.2f\n", a, b, (float) a / b); 
12    printf("% d %% % d  = %d\n", a, b, a % b); 
13    return 0; 
14 }
